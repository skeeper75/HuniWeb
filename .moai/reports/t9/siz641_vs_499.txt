$ psql <URL> -c "
SELECT a.mat_cd, m.mat_nm, a.unit_price AS p_480x320, b.unit_price AS p_316x467,
       round(a.unit_price - b.unit_price, 2) AS diff,
       round(b.unit_price * 153600.0/147572.0, 2) AS area_scaled_expect
FROM t_prc_component_prices a
LEFT JOIN t_prc_component_prices b
  ON b.comp_cd='COMP_PAPER' AND b.plt_siz_cd='SIZ_000499'
 AND b.mat_cd=a.mat_cd AND b.min_qty=a.min_qty
LEFT JOIN t_mat_materials m ON m.mat_cd=a.mat_cd
WHERE a.comp_cd='COMP_PAPER' AND a.plt_siz_cd='SIZ_000641'
ORDER BY a.mat_cd;
"
