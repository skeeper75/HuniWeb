SELECT cp.plt_siz_cd, z.siz_nm, (z.work_width*z.work_height)::int AS area,
       count(*) AS n_rows, min(cp.unit_price) AS min_p, max(cp.unit_price) AS max_p,
       min(cp.reg_dt)::date AS first_reg
FROM t_prc_component_prices cp
LEFT JOIN t_siz_sizes z ON z.siz_cd=cp.plt_siz_cd
WHERE cp.comp_cd='COMP_PAPER' AND cp.plt_siz_cd IS NOT NULL
GROUP BY cp.plt_siz_cd, z.siz_nm, z.work_width, z.work_height
ORDER BY area NULLS LAST;
