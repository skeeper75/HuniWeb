$ psql <URL> -c "
SELECT cp.mat_cd, m.mat_nm, cp.plt_siz_cd, z.siz_nm,
       (z.work_width*z.work_height)::int AS area,
       cp.unit_price, cp.min_qty, cp.apply_ymd, cp.reg_dt
FROM t_prc_component_prices cp
LEFT JOIN t_siz_sizes z ON z.siz_cd=cp.plt_siz_cd
LEFT JOIN t_mat_materials m ON m.mat_cd=cp.mat_cd
WHERE cp.comp_cd='COMP_PAPER'
  AND cp.plt_siz_cd IN ('SIZ_000499','SIZ_000640','SIZ_000641')
  AND cp.mat_cd IN ('MAT_000074','MAT_000078')
ORDER BY cp.mat_cd, cp.plt_siz_cd, cp.min_qty;
"
